    SORT gt_venexpiryinfo BY lifnr ASCENDING seq DESCENDING.
    DELETE ADJACENT DUPLICATES FROM gt_venexpiryinfo COMPARING lifnr.   " Anil
  ENDIF.

  SELECT * FROM zdepartment_id
    INTO CORRESPONDING FIELDS OF TABLE gt_departmentinfo.
  IF sy-subrc IS INITIAL.
    SORT gt_departmentinfo BY dept.
  ENDIF.

  SELECT a~partner, a~bpkind, b~text40 FROM but000 AS a
    INNER JOIN tb004t AS b
    ON a~bpkind = b~bpkind
    INTO CORRESPONDING FIELDS OF TABLE @gt_vendetails
    FOR ALL ENTRIES IN @gt_crcomptaxinfo
    WHERE a~partner = @gt_crcomptaxinfo-zvencode AND b~spras = 'E'.

    SORT gt_vendetails BY partner.
    DELETE ADJACENT DUPLICATES FROM gt_vendetails COMPARING partner.      " Commited by Anil
  ENDIF.

    INNER JOIN adr6 AS b

      IF p_chk14 IS NOT INITIAL.

          date      = sy-datum

        INTO @DATA(ls_ekko)

          ENDIF.
          IF lv_days_diff = p_13c.
            gwa_notifications-evastatus = 'Qualified'.
            gwa_notifications-reminderdays = p_13c.
            IF p_chk13c IS NOT INITIAL.
              gwa_notifications-evaexpremnotif = 'X'.
              CONCATENATE lv_notifinfogrp3 TEXT-aw3 p_13c ';' INTO lv_notifinfogrp3 SEPARATED BY space.

          ENDIF.

            gwa_notifications-reminderdays = p_13b.
            IF p_chk13b IS NOT INITIAL.
              gwa_notifications-evaexpremnotif = 'X'.
              CONCATENATE lv_notifinfogrp3 TEXT-aw3 p_13b ';' INTO lv_notifinfogrp3 SEPARATED BY space.
            ENDIF.
          ENDIF.
          IF lv_days_diff = p_13a.
            gwa_notifications-evastatus = 'Qualified'.
            gwa_notifications-reminderdays = p_13a.
            IF p_chk13a IS NOT INITIAL.
              gwa_notifications-evaexpfirnotif = 'X'.
              CONCATENATE lv_notifinfogrp3 TEXT-aw3 p_13a ';' INTO lv_notifinfogrp3 SEPARATED BY space.
            ENDIF.
          ENDIF.
