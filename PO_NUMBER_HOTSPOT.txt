                             text    TYPE scrtext_l.

  wa_fieldcat-tabname    = 'T_FINAL'.
  wa_fieldcat-fieldname  = field.
  wa_fieldcat-scrtext_l  = text.

  IF field EQ 'TXZ01'.
    wa_fieldcat-lowercase = 'X'.
    wa_fieldcat-outputlen = 20.
  ENDIF.

  IF field EQ 'LBLNI'.
    wa_fieldcat-convexit = 'ALPHA'.
  ENDIF.

  IF  field EQ 'EBELN'.
    wa_fieldcat-convexit = 'ALPHA'.
    wa_fieldcat-hotspot = 'X'.
  ENDIF.

  IF field EQ 'APPROVED'.
    wa_fieldcat-checkbox = 'X'.
  ENDIF.

  IF field EQ 'APPROVED_AT1' OR field EQ 'APPROVED_AT2' OR field EQ 'APPROVED_AT3' OR field EQ 'APPROVED_AT4' OR field EQ 'APPROVED_AT5'.
    wa_fieldcat-no_zero = 'X'.
  ENDIF.

  APPEND wa_fieldcat TO t_fieldcat.

ENDFORM.                    " FORM_FIELDCAT

FORM form_user_command USING r_ucomm LIKE sy-ucomm
                             rs_selfield TYPE slis_selfield.

* Local data declaration
  DATA: lv_ref1  TYPE REF TO cl_gui_alv_grid.

  CALL FUNCTION 'GET_GLOBALS_FROM_SLVC_FULLSCR'
    IMPORTING
      e_grid = lv_ref1.
  CALL METHOD lv_ref1->check_changed_data.

  CASE r_ucomm.

    WHEN 'APPROVE'.
      PERFORM form_action_ses USING 'A'.

    WHEN 'REJECT'.
      PERFORM form_action_ses USING 'R'.

    WHEN '&IC1'.
    AUTHORITY-CHECK OBJECT 'ZAUTNO'
   ID 'ACTVT' FIELD '03'.
    IF sy-subrc NE 0.
      MESSAGE i010(zmm) WITH 'No Authorization to edit'.
    ELSE.
      SET PARAMETER ID 'BES' FIELD rs_selfield-value.
       CALL TRANSACTION 'ME23N'.
    ENDIF.
  ENDCASE.

  rs_selfield-refresh = 'X'.

ENDFORM.                    "FORM_USER_COMMAND
