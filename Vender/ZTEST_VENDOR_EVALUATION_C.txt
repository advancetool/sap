FORM fcat1  USING    VALUE(p_1)
                    VALUE(p_2)
                    VALUE(p_3).


  DATA: lv_is_key TYPE abap_bool VALUE abap_false,
        lv_is_cmt TYPE abap_bool VALUE abap_false.

  " Set lowercase and tabname
  wa_fcat-lowercase = 'X'.
  wa_fcat-tabname = 'it_dat1'.
  ls_fcat-lowercase = 'X'.
  ls_fcat-tabname = 'it_dat1'.

  IF p_1 = 'LIFNR' OR
        p_1 = 'NAME1' .
    ls_fcat-key = abap_true.
    wa_fcat-key = abap_true.
  ENDIF.

  IF p_1 CS 'CMT'.
    ls_fcat-outputlen = 30.
    wa_fcat-outputlen = 30.
  ENDIF.

  " Populate wa_fcat fields
  wa_fcat-fieldname = p_1.
  wa_fcat-edit = p_2.
  wa_fcat-seltext_l = p_3.

  IF wa_fcat-fieldname = 'CHK'.
    wa_fcat-checkbox = abap_true.
    wa_fcat-edit = abap_true.
  ELSE.
    wa_fcat-checkbox = abap_false.
  ENDIF.

  APPEND wa_fcat TO it_fcat.
  CLEAR wa_fcat.

  IF p_arch IS INITIAL AND ( p_1 = 'SEQ' OR p_1 = 'AIND' OR p_1 = 'ADT' ).
    DELETE it_fcat WHERE fieldname = p_1.
  ENDIF.

ENDFORM.
